import React, { useEffect, useMemo, useState } from 'react';
import {
  Area, AreaChart, Bar, BarChart, CartesianGrid, ResponsiveContainer,
  Tooltip, XAxis, YAxis
} from 'recharts';
import './App.css';

const IMAGES = {
  patient: 'https://images.unsplash.com/photo-1494790108377-be9c29b29330?w=160&auto=format&fit=crop&q=85',
  doctor: 'https://images.unsplash.com/photo-1559839734-2b71ea197ec2?w=160&auto=format&fit=crop&q=85'
};

const INITIAL_USERS = [
  {
    email: 'maria@nutrisalud.com',
    password: '123',
    role: 'cliente',
    nombre: 'María Pérez',
    edad: 28,
    estatura: 1.65,
    peso: 65.5
  },
  {
    email: 'carlos@nutrisalud.com',
    password: '123',
    role: 'cliente',
    nombre: 'Carlos Gómez',
    edad: 35,
    estatura: 1.75,
    peso: 78
  },
  {
    email: 'doctora@nutrisalud.com',
    password: 'admin',
    role: 'admin',
    nombre: 'Dra. Valeria Martínez',
    especialidad: 'Nutrióloga clínica'
  }
];

const INITIAL_SURVEY = [
  { name: 'Verduras', porciones: 5 },
  { name: 'Frutas', porciones: 4 },
  { name: 'Cereales', porciones: 3 },
  { name: 'Proteína', porciones: 6 }
];

const readStorage = (key, fallback) => {
  try {
    const value = localStorage.getItem(key);
    return value ? JSON.parse(value) : fallback;
  } catch {
    return fallback;
  }
};

function App() {
  const [users, setUsers] = useState(() =>
    readStorage('nutri_users', INITIAL_USERS)
  );

  const [currentUser, setCurrentUser] = useState(() =>
    readStorage('nutri_session', null)
  );

  const [activeTab, setActiveTab] = useState('inicio');
  const [authEmail, setAuthEmail] = useState('');
  const [authPassword, setAuthPassword] = useState('');
  const [authError, setAuthError] = useState('');
  const [showPassword, setShowPassword] = useState(false);

  const [surveyData, setSurveyData] = useState(() =>
    readStorage('nutri_survey', INITIAL_SURVEY)
  );

  const [bpDraft, setBpDraft] = useState({
    PAS: '',
    PAD: '',
    fecha: new Date().toISOString().slice(0, 10),
    hora: new Date().toTimeString().slice(0, 5),
    notas: ''
  });

  const [bpNotice, setBpNotice] = useState('');
  const [activeChatClient, setActiveChatClient] = useState('maria@nutrisalud.com');
  const [inputMsg, setInputMsg] = useState('');
  const [selectedProfileEmail, setSelectedProfileEmail] = useState('maria@nutrisalud.com');
  const [profileNotice, setProfileNotice] = useState('');
  const [passwordNotice, setPasswordNotice] = useState('');

  const [currentPassword, setCurrentPassword] = useState('');
  const [newPassword, setNewPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  const [profileDraft, setProfileDraft] = useState({});

  const [showNewPatient, setShowNewPatient] = useState(false);
  const [newPatient, setNewPatient] = useState({
    nombre: '',
    email: '',
    password: ''
  });

  const [newPatientNotice, setNewPatientNotice] = useState('');

  const [conversations, setConversations] = useState(() =>
    readStorage('nutri_chats', {
      'maria@nutrisalud.com': [
        {
          sender: 'admin',
          text: '¡Hola María! Bienvenida a tu espacio de seguimiento. ¿Cómo te has sentido esta semana?'
        },
        {
          sender: 'cliente',
          text: 'Hola doctora, tengo una duda sobre mi seguimiento.'
        }
      ]
    })
  );

  const isClient = currentUser?.role === 'cliente';

  const patients = useMemo(
    () => users.filter(user => user.role === 'cliente'),
    [users]
  );

  const selectedChat = isClient ? currentUser?.email : activeChatClient;
  const currentMessages = conversations[selectedChat] || [];

  const profileTarget = isClient
    ? currentUser
    : (
      patients.find(patient => patient.email === selectedProfileEmail)
      || patients[0]
    );

  const pressureTarget = isClient
    ? currentUser
    : (
      patients.find(patient => patient.email === selectedProfileEmail)
      || patients[0]
    );

  const pressureRecords = pressureTarget?.pressureHistory || [];
  const latestPressure = pressureRecords[0] || null;

  const pressureChartData = [...pressureRecords]
    .reverse()
    .map(record => ({ ...record, day: record.fecha }));

  useEffect(() => {
    if (profileTarget) {
      setProfileDraft({
        nombre: '',
        fechaNacimiento: '',
        edad: '',
        sexo: '',
        estatura: '',
        peso: '',
        foto: '',
        email: '',
        telefono: '',
        direccion: '',
        contactoEmergencia: '',
        antecedentes: '',
        alergias: '',
        medicamentos: '',
        restricciones: '',
        objetivo: '',
        ...profileTarget
      });
    }
  }, [profileTarget?.email, currentUser]);

  useEffect(() => {
    localStorage.setItem('nutri_users', JSON.stringify(users));
  }, [users]);

  useEffect(() => {
    localStorage.setItem('nutri_session', JSON.stringify(currentUser));
  }, [currentUser]);

  useEffect(() => {
    localStorage.setItem('nutri_chats', JSON.stringify(conversations));
  }, [conversations]);

  useEffect(() => {
    localStorage.setItem('nutri_survey', JSON.stringify(surveyData));
  }, [surveyData]);

  const login = (event) => {
    event.preventDefault();

    const found = users.find(user =>
      user.email.toLowerCase() === authEmail.trim().toLowerCase()
      && user.password === authPassword
    );

    if (!found) {
      setAuthError('No pudimos validar tus datos. Revisa tu correo y contraseña.');
      return;
    }

    setCurrentUser(found);
    setActiveTab(found.role === 'admin' ? 'admin-dashboard' : 'inicio');
    setAuthError('');
  };

  const logout = () => {
    setCurrentUser(null);
    setAuthPassword('');
    setActiveTab('inicio');
  };

  const sendMessage = (event) => {
    event.preventDefault();

    const message = inputMsg.trim();

    if (!message || !selectedChat) return;

    const sender = isClient ? 'cliente' : 'admin';

    setConversations(previous => ({
      ...previous,
      [selectedChat]: [
        ...(previous[selectedChat] || []),
        {
          sender,
          text: message,
          time: new Date().toLocaleTimeString('es-MX', {
            hour: '2-digit',
            minute: '2-digit'
          })
        }
      ]
    }));

    setInputMsg('');
  };

  const saveProfile = (event) => {
    event.preventDefault();

    if (!profileTarget) return;

    const updatedUser = {
      ...profileTarget,
      ...profileDraft,
      email: profileTarget.email
    };

    setUsers(previous =>
      previous.map(user =>
        user.email === profileTarget.email ? updatedUser : user
      )
    );

    if (isClient) {
      setCurrentUser(updatedUser);
    }

    setProfileNotice(
      isClient
        ? 'Tu información se guardó correctamente.'
        : 'El expediente del paciente se actualizó correctamente.'
    );
  };

  const createPatient = (event) => {
    event.preventDefault();

    const email = newPatient.email.trim().toLowerCase();

    if (users.some(user => user.email.toLowerCase() === email)) {
      setNewPatientNotice('Ya existe una cuenta con ese correo electrónico.');
      return;
    }

    const patient = {
      email,
      password: newPatient.password,
      role: 'cliente',
      nombre: newPatient.nombre.trim(),
      fechaNacimiento: '',
      edad: '',
      sexo: '',
      estatura: '',
      peso: '',
      telefono: '',
      direccion: '',
      contactoEmergencia: '',
      antecedentes: '',
      alergias: '',
      medicamentos: '',
      restricciones: '',
      objetivo: ''
    };

    setUsers(previous => [...previous, patient]);
    setSelectedProfileEmail(email);
    setActiveChatClient(email);

    setConversations(previous => ({
      ...previous,
      [email]: []
    }));

    setNewPatient({
      nombre: '',
      email: '',
      password: ''
    });

    setNewPatientNotice('');
    setShowNewPatient(false);
    setActiveTab('perfil');

    setProfileNotice(
      'Perfil creado. El paciente ya puede iniciar sesión y completar sus datos.'
    );
  };

  const changePassword = (event) => {
    event.preventDefault();
    setPasswordNotice('');

    if (currentPassword !== currentUser.password) {
      setPasswordNotice('La contraseña actual no es correcta.');
      return;
    }

    if (newPassword.length < 6) {
      setPasswordNotice('La nueva contraseña debe tener al menos 6 caracteres.');
      return;
    }

    if (newPassword !== confirmPassword) {
      setPasswordNotice('La confirmación no coincide con la nueva contraseña.');
      return;
    }

    const updatedUser = {
      ...currentUser,
      password: newPassword
    };

    setUsers(previous =>
      previous.map(user =>
        user.email === currentUser.email ? updatedUser : user
      )
    );

    setCurrentUser(updatedUser);
    setCurrentPassword('');
    setNewPassword('');
    setConfirmPassword('');

    setPasswordNotice('Tu contraseña se actualizó correctamente.');
  };

  const classifyPressure = (PAS, PAD) => {
    const systolic = Number(PAS);
    const diastolic = Number(PAD);

    if (
      !Number.isFinite(systolic)
      || !Number.isFinite(diastolic)
      || systolic <= 0
      || diastolic <= 0
    ) {
      return {
        label: 'Sin clasificación',
        level: 'neutral',
        advice: 'Registra una medición válida.'
      };
    }

    if (systolic >= 140 || diastolic >= 90) {
      return {
        label: 'Hipertensión · Estadio 2',
        level: 'danger',
        advice: 'Busca valoración de un profesional de salud. Si tienes síntomas intensos o la lectura es muy alta, solicita atención urgente.'
      };
    }

    if (
      systolic >= 130
      || (diastolic >= 80 && diastolic <= 89)
    ) {
      return {
        label: 'Hipertensión · Estadio 1',
        level: 'warning',
        advice: 'Conviene consultar con un profesional de salud y dar seguimiento a tus mediciones.'
      };
    }

    if (
      systolic >= 120
      && systolic <= 129
      && diastolic < 80
    ) {
      return {
        label: 'Elevada',
        level: 'warning',
        advice: 'La presión está elevada. Mantén seguimiento y consulta cómo mejorar tus hábitos.'
      };
    }

    if (systolic < 120 && diastolic < 80) {
      return {
        label: 'Normal',
        level: 'good',
        advice: 'La lectura está dentro del rango normal de la tabla.'
      };
    }

    return {
      label: 'Fuera de los rangos de la tabla',
      level: 'neutral',
      advice: 'Repite la medición en reposo y consulta a un profesional si persiste.'
    };
  };

  const savePressure = (event) => {
    event.preventDefault();

    if (!pressureTarget) return;

    const PAS = Number(bpDraft.PAS);
    const PAD = Number(bpDraft.PAD);

    if (
      !Number.isFinite(PAS)
      || !Number.isFinite(PAD)
      || PAS < 50
      || PAS > 260
      || PAD < 30
      || PAD > 160
    ) {
      setBpNotice('Ingresa valores válidos para la presión sistólica y diastólica.');
      return;
    }

    const record = {
      id: Date.now(),
      PAS,
      PAD,
      fecha: bpDraft.fecha,
      hora: bpDraft.hora,
      notas: bpDraft.notas.trim()
    };

    const updatedUser = {
      ...pressureTarget,
      pressureHistory: [
        record,
        ...(pressureTarget.pressureHistory || [])
      ]
    };

    setUsers(previous =>
      previous.map(user =>
        user.email === pressureTarget.email ? updatedUser : user
      )
    );

    if (isClient) {
      setCurrentUser(updatedUser);
    }

    setBpDraft({
      PAS: '',
      PAD: '',
      fecha: new Date().toISOString().slice(0, 10),
      hora: new Date().toTimeString().slice(0, 5),
      notas: ''
    });

    setBpNotice('Medición guardada correctamente.');
  };

  const updateProfilePhoto = (file) => {
    if (!file || !profileTarget) return;

    if (!file.type.startsWith('image/')) {
      setProfileNotice('Selecciona un archivo de imagen válido.');
      return;
    }

    if (file.size > 2 * 1024 * 1024) {
      setProfileNotice('La fotografía debe pesar menos de 2 MB.');
      return;
    }

    const reader = new FileReader();

    reader.onload = () => {
      setProfileDraft(previous => ({
        ...previous,
        foto: reader.result
      }));
    };

    reader.readAsDataURL(file);
    setProfileNotice('');
  };

  const updatePortions = (index, value) => {
    const safeValue = Math.max(
      0,
      Math.min(30, Number(value) || 0)
    );

    setSurveyData(previous =>
      previous.map((item, i) =>
        i === index
          ? { ...item, porciones: safeValue }
          : item
      )
    );
  };

  const menu = isClient
    ? [
      ['inicio', '▦', 'Inicio'],
      ['presion', '♡', 'Presión arterial'],
      ['encuesta', '▥', 'Mi seguimiento'],
      ['mensajes', '✉', 'Mensajes'],
      ['perfil', '◉', 'Mi perfil']
    ]
    : [
      ['admin-dashboard', '▦', 'Panel general'],
      ['presion', '♡', 'Presión arterial'],
      ['mensajes', '✉', 'Mensajes'],
      ['perfil', '◉', 'Perfiles']
    ];

  if (!currentUser) {
    return (
      <div className="login-page">
        <div className="login-visual">
          <div className="login-brand">
            <span className="brand-mark">n+</span>
            <span>NutriSalud<span className="brand-plus">+</span></span>
          </div>

          <div className="login-visual-copy">
            <span className="eyebrow light">
              BIENESTAR CON ACOMPAÑAMIENTO
            </span>
            <h1>Pequeños cambios.<br />Grandes resultados.</h1>
            <p>
              Un espacio pensado para acompañarte en cada paso de tu bienestar.
            </p>
          </div>

          <div className="login-visual-footer">
            Nutrición personalizada · Seguimiento profesional
          </div>
        </div>

        <div className="login-panel">
          <div className="login-form-wrap">
            <div className="mobile-brand">
              <span className="brand-mark">n+</span>
              <strong>NutriSalud<span className="brand-plus">+</span></strong>
            </div>

            <span className="eyebrow">PORTAL DE BIENESTAR</span>
            <h2>Qué gusto tenerte aquí</h2>
            <p className="muted">
              Inicia sesión para consultar tu información y continuar tu seguimiento.
            </p>

            {authError && (
              <div className="alert-error" role="alert">
                {authError}
              </div>
            )}

            <form className="login-form" onSubmit={login}>
              <label htmlFor="email">Correo electrónico</label>
              <input
                id="email"
                type="email"
                autoComplete="username"
                placeholder="tu@correo.com"
                value={authEmail}
                onChange={e => setAuthEmail(e.target.value)}
                required
              />

              <label htmlFor="password">Contraseña</label>

              <div className="password-field">
                <input
                  id="password"
                  type={showPassword ? 'text' : 'password'}
                  autoComplete="current-password"
                  placeholder="Ingresa tu contraseña"
                  value={authPassword}
                  onChange={e => setAuthPassword(e.target.value)}
                  required
                />

                <button
                  type="button"
                  className="password-toggle"
                  onClick={() => setShowPassword(value => !value)}
                >
                  {showPassword ? 'Ocultar' : 'Mostrar'}
                </button>
              </div>

              <button type="submit" className="button button-primary login-submit">
                Iniciar sesión <span>→</span>
              </button>
            </form>

            <div className="demo-access">
              <strong>Accesos de demostración</strong>
              <p><span>Paciente</span> maria@nutrisalud.com <b>·</b> 123</p>
              <p><span>Profesional</span> doctora@nutrisalud.com <b>·</b> admin</p>
            </div>

            <p className="login-privacy">
              Este portal es una demostración. No ingreses información médica real.
            </p>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="app-shell">
      <aside className="sidebar">
        <div>
          <div className="sidebar-brand">
            <span className="brand-mark">n+</span>
            <div>
              <strong>NutriSalud<span className="brand-plus">+</span></strong>
              <small>Tu espacio de bienestar</small>
            </div>
          </div>

          <div className="nav-caption">MENÚ PRINCIPAL</div>

          <nav className="sidebar-nav" aria-label="Navegación principal">
            {menu.map(([id, icon, label]) => (
              <button
                key={id}
                className={`nav-item ${activeTab === id ? 'active' : ''}`}
                onClick={() => setActiveTab(id)}
              >
                <span className="nav-icon">{icon}</span>
                <span>{label}</span>
                {activeTab === id && <span className="nav-indicator" />}
              </button>
            ))}
          </nav>
        </div>

        <div className="sidebar-bottom">
          <div className="help-card">
            <span className="help-icon">♡</span>
            <strong>Estamos para ayudarte</strong>
            <p>Si tienes alguna duda, escríbenos desde Mensajes.</p>
            <button onClick={() => setActiveTab('mensajes')}>
              Contactar <span>→</span>
            </button>
          </div>

          <div className="profile-mini">
            <img
              src={currentUser.foto || (isClient ? IMAGES.patient : IMAGES.doctor)}
              alt=""
            />
            <div className="profile-copy">
              <strong>{currentUser.nombre}</strong>
              <small>
                {isClient
                  ? 'Paciente'
                  : currentUser.especialidad || 'Profesional'}
              </small>
            </div>
            <button
              className="logout-icon"
              onClick={logout}
              title="Cerrar sesión"
              aria-label="Cerrar sesión"
            >
              ↗
            </button>
          </div>
        </div>
      </aside>

      <main className="main-area">
        <header className="topbar">
          <div>
            <span className="topbar-label">
              NUTRISALUD+ / {isClient ? 'PORTAL DEL PACIENTE' : 'ÁREA PROFESIONAL'}
            </span>
            <div className="topbar-date">
              {new Date().toLocaleDateString('es-MX', {
                weekday: 'long',
                day: 'numeric',
                month: 'long',
                year: 'numeric'
              })}
            </div>
          </div>

          <div className="topbar-right">
            <span className="secure-label">
              <span className="secure-dot" /> Espacio de demostración
            </span>
            <div className="topbar-avatar">
              {currentUser.foto
                ? <img src={currentUser.foto} alt="Fotografía de perfil" />
                : currentUser.nombre.split(' ').map(part => part[0]).slice(0, 2).join('')}
            </div>
          </div>
        </header>

        <div className="page-content">
          {isClient && activeTab === 'inicio' && (
            <>
              <section className="page-heading">
                <div>
                  <span className="eyebrow">INICIO · TU ESTADO GENERAL</span>
                  <h1>
                    Hola, {currentUser.nombre.split(' ')[0]} <span className="wave">✳</span>
                  </h1>
                  <p>
                    Consulta de un vistazo tus datos registrados y el seguimiento de tu bienestar.
                  </p>
                </div>

                <button
                  className="button button-primary"
                  onClick={() => setActiveTab('encuesta')}
                >
                  Actualizar seguimiento <span>→</span>
                </button>
              </section>

              <div className="section-title-row">
                <div>
                  <h2>Estado general</h2>
                  <p>Información que has registrado en tu perfil</p>
                </div>
                <span className="subtle-tag">Datos personales</span>
              </div>

              <section className="metrics-grid">
                <article className="metric-card">
                  <div className="metric-top">
                    <span className="metric-icon mint">↗</span>
                    <span className="metric-label">Peso registrado</span>
                  </div>
                  <div className="metric-value">
                    {currentUser.peso || 65.5}<small> kg</small>
                  </div>
                  <p className="metric-foot">Último dato guardado</p>
                </article>

                <article className="metric-card">
                  <div className="metric-top">
                    <span className="metric-icon lavender">⌁</span>
                    <span className="metric-label">Estatura</span>
                  </div>
                  <div className="metric-value">
                    {currentUser.estatura || 1.65}<small> m</small>
                  </div>
                  <p className="metric-foot">Información de perfil</p>
                </article>

                <article className="metric-card">
                  <div className="metric-top">
                    <span className="metric-icon peach">♡</span>
                    <span className="metric-label">Edad</span>
                  </div>
                  <div className="metric-value">
                    {currentUser.edad || '—'}<small> años</small>
                  </div>
                  <p className="metric-foot">Información de perfil</p>
                </article>
              </section>

              <section className="surface-card home-pressure-summary">
                <div>
                  <span className="eyebrow">PRESIÓN ARTERIAL</span>
                  <h3>
                    {latestPressure
                      ? `${latestPressure.PAS}/${latestPressure.PAD} mmHg`
                      : 'Sin mediciones registradas'}
                  </h3>
                  <p>
                    {latestPressure
                      ? classifyPressure(latestPressure.PAS, latestPressure.PAD).label
                      : 'Agrega tu primera medición para conocer tu clasificación.'}
                  </p>
                </div>

                <button
                  className="button button-primary"
                  onClick={() => setActiveTab('presion')}
                >
                  Consultar presión arterial →
                </button>
              </section>

              <section className="surface-card home-next-step">
                <div>
                  <span className="eyebrow">TU ACOMPAÑAMIENTO</span>
                  <h3>¿Tienes alguna duda sobre tu seguimiento?</h3>
                  <p>Comunícate con tu profesional para resolver tus preguntas.</p>
                </div>
                <button
                  className="text-button"
                  onClick={() => setActiveTab('mensajes')}
                >
                  Ir a mensajes →
                </button>
              </section>
            </>
          )}

          {activeTab === 'presion' && (
            <>
              <PageTitle
                eyebrow="SEGUIMIENTO DE SALUD"
                title="Presión arterial"
                description="Registra tus mediciones y consulta cómo se clasifican según los rangos que compartiste."
              />

              {!isClient && patients.length > 0 && (
                <section className="surface-card profile-selector">
                  <label htmlFor="pressure-patient-select">Paciente</label>
                  <select
                    id="pressure-patient-select"
                    value={pressureTarget?.email || ''}
                    onChange={e => {
                      setSelectedProfileEmail(e.target.value);
                      setBpNotice('');
                    }}
                  >
                    {patients.map(patient => (
                      <option key={patient.email} value={patient.email}>
                        {patient.nombre} · {patient.email}
                      </option>
                    ))}
                  </select>
                </section>
              )}

              {pressureTarget ? (
                <>
                  <section className="pressure-overview-grid">
                    <article className="surface-card pressure-current-card">
                      <span className="eyebrow">ÚLTIMA MEDICIÓN</span>

                      {latestPressure ? (
                        <>
                          <div className="pressure-reading">
                            {latestPressure.PAS}
                            <small> / {latestPressure.PAD}</small>
                          </div>
                          <span className="pressure-unit">mmHg</span>

                          <span className={`pressure-status ${classifyPressure(latestPressure.PAS, latestPressure.PAD).level}`}>
                            {classifyPressure(latestPressure.PAS, latestPressure.PAD).label}
                          </span>

                          <p>
                            {classifyPressure(latestPressure.PAS, latestPressure.PAD).advice}
                          </p>

                          <small>
                            Registrada: {latestPressure.fecha} · {latestPressure.hora}
                          </small>
                        </>
                      ) : (

                      Enviar <span>↑</span>
                    </button>
                  </form>
                </div>
              </section>
            </>
          )}

          {!isClient && activeTab === 'admin-dashboard' && (
            <>
              <PageTitle
                eyebrow="ÁREA PROFESIONAL"
                title={`Buen día, ${currentUser.nombre.split(' ')[0]}`}
                description="Aquí tienes un resumen de la actividad de tu consulta."
              />

              <section className="admin-stats">
                <article className="metric-card">
                  <div className="metric-top">
                    <span className="metric-icon mint">♙</span>
                    <span className="metric-label">Pacientes registrados</span>
                  </div>
                  <div className="metric-value">
                    {patients.length.toString().padStart(2, '0')}
                  </div>
                  <p className="metric-foot">Perfiles de demostración</p>
                </article>

                <article className="metric-card">
                  <div className="metric-top">
                    <span className="metric-icon lavender">✉</span>
                    <span className="metric-label">Conversaciones</span>
                  </div>
                  <div className="metric-value">
                    {Object.keys(conversations).length.toString().padStart(2, '0')}
                  </div>
                  <p className="metric-foot">Chats disponibles</p>
                </article>

                <article className="metric-card">
                  <div className="metric-top">
                    <span className="metric-icon peach">♡</span>
                    <span className="metric-label">Especialidad</span>
                  </div>
                  <div className="metric-value text-value">Nutrición</div>
                  <p className="metric-foot">Atención y seguimiento</p>
                </article>
              </section>

              <section className="surface-card patient-list-card">
                <div className="card-heading">
                  <div>
                    <h3>Directorio de pacientes</h3>
                    <p>Administra los perfiles individuales de tu consulta.</p>
                  </div>

                  <div className="directory-actions">
                    <span className="soft-pill">{patients.length} perfiles</span>
                    <button
                      className="button button-primary"
                      type="button"
                      onClick={() => {
                        setShowNewPatient(value => !value);
                        setNewPatientNotice('');
                      }}
                    >
                      ＋ Nuevo paciente
                    </button>
                  </div>
                </div>

                {showNewPatient && (
                  <form className="new-patient-form" onSubmit={createPatient}>
                    <label>
                      Nombre completo
                      <input
                        value={newPatient.nombre}
                        onChange={e => setNewPatient({
                          ...newPatient,
                          nombre: e.target.value
                        })}
                        placeholder="Nombre del paciente"
                        required
                      />
                    </label>

                    <label>
                      Correo electrónico
                      <input
                        type="email"
                        value={newPatient.email}
                        onChange={e => setNewPatient({
                          ...newPatient,
                          email: e.target.value
                        })}
                        placeholder="paciente@correo.com"
                        required
                      />
                    </label>

                    <label>
                      Contraseña temporal
                      <input
                        type="password"
                        minLength="6"
                        value={newPatient.password}
                        onChange={e => setNewPatient({
                          ...newPatient,
                          password: e.target.value
                        })}
                        placeholder="Mínimo 6 caracteres"
                        required
                      />
                    </label>

                    <button className="button button-primary" type="submit">
                      Crear perfil
                    </button>

                    {newPatientNotice && (
                      <p className="form-feedback error-feedback">
                        {newPatientNotice}
                      </p>
                    )}
                  </form>
                )}

                <div className="patient-table-wrap">
                  <table className="patient-table">
                    <thead>
                      <tr>
                        <th>Paciente</th>
                        <th>Correo electrónico</th>
                        <th>Edad</th>
                        <th>Peso</th>
                        <th>Acción</th>
                      </tr>
                    </thead>

                    <tbody>
                      {patients.map(patient => (
                        <tr key={patient.email}>
                          <td><strong>{patient.nombre}</strong></td>
                          <td>{patient.email}</td>
                          <td>{patient.edad || '—'} años</td>
                          <td>{patient.peso ? `${patient.peso} kg` : '—'}</td>
                          <td>
                            <button
                              className="table-action"
                              onClick={() => {
                                setSelectedProfileEmail(patient.email);
                                setActiveTab('perfil');
                              }}
                            >
                              Ver perfil →
                            </button>

                            <button
                              className="table-action"
                              onClick={() => {
                                setActiveChatClient(patient.email);
                                setActiveTab('mensajes');
                              }}
                            >
                              Mensaje
                            </button>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </section>

              <div className="notice-banner">
                <span>i</span>
                <p>
                  Los perfiles, métricas y conversaciones mostrados son datos de demostración.
                  Para uso clínico real se necesita autenticación segura, permisos y almacenamiento protegido.
                </p>
              </div>
            </>
          )}
        </div>

        <footer className="app-footer">
          <span>© {new Date().getFullYear()} NutriSalud+</span>
          <span>Un acompañamiento para cada etapa de tu bienestar.</span>
        </footer>
      </main>
    </div>
  );
}

function PageTitle({ eyebrow, title, description }) {
  return (
    <section className="page-heading simple-heading">
      <div>
        <span className="eyebrow">{eyebrow}</span>
        <h1>{title}</h1>
        <p>{description}</p>
      </div>
    </section>
  );
}
